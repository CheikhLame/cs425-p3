## Analysis

The subnet calculations use fixed-width bit operations, so masking, network
and broadcast address calculation, host counting, and containment checks all
run in constant time. Splitting a network takes time proportional to the
number of subnets printed. The forwarding table uses a linear scan: adding a
route checks existing entries for duplicates, and lookup checks each route
for a match, so both operations take O(R) time for R routes. Loading R routes
therefore takes O(R^2) time in the worst case and O(R) memory. This is
appropriate for the assignment's sample tables; a binary trie would be a
better fit for a large routing table.

The tests cover strict rejection of malformed addresses and routes, CIDR
boundary cases including /0, /31, and /32, subnet splitting, table loading,
and longest-prefix selection independent of insertion order. The latest
coverage report for `src/lab.c` shows 100% line and branch coverage, and all
six test cases pass. The only excluded branches are failures from allocation
and stream-status library calls.

bool ip_parse(const char *text, uint32_t *addr)
{
  if (text == NULL || addr == NULL)
  {
    return false;
  }

  const unsigned char *cursor = (const unsigned char *) text;
  uint32_t parsed = 0;

  for (unsigned int octet = 0; octet < 4; octet++)
  {
    if (*cursor < '0' || *cursor > '9')
    {
      return false;
    }

    if (*cursor == '0' && cursor[1] >= '0' && cursor[1] <= '9')
    {
      return false;
    }

    unsigned int value = 0;
    while (*cursor >= '0' && *cursor <= '9')
    {
      value = value * 10U + (unsigned int) (*cursor - '0');
      if (value > 255U)
      {
        return false;
      }
      cursor++;
    }

    parsed = (parsed << 8) | (uint32_t) value;
    if (octet < 3)
    {
      if (*cursor != '.')
      {
        return false;
      }
      cursor++;
    }
    else if (*cursor != '\0')
    {
      return false;
    }
  }

  *addr = parsed;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
  (void) snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
                  (unsigned int) ((addr >> 24) & UINT32_C(0xff)),
                  (unsigned int) ((addr >> 16) & UINT32_C(0xff)),
                  (unsigned int) ((addr >> 8) & UINT32_C(0xff)),
                  (unsigned int) (addr & UINT32_C(0xff)));
}

bool cidr_parse(const char *text, cidr_t *block)
{
  if (text == NULL || block == NULL)
  {
    return false;
  }

  const char *slash = text;
  while (*slash != '\0' && *slash != '/')
  {
    slash++;
  }
  if (*slash != '/')
  {
    return false;
  }

  size_t address_length = (size_t) (slash - text);
  if (address_length == 0 || address_length >= IP_STR_LEN)
  {
    return false;
  }

  char address[IP_STR_LEN];
  for (size_t i = 0; i < address_length; i++)
  {
    address[i] = text[i];
  }
  address[address_length] = '\0';

  const unsigned char *cursor = (const unsigned char *) (slash + 1);
  if (*cursor < '0' || *cursor > '9')
  {
    return false;
  }
  if (*cursor == '0' && cursor[1] >= '0' && cursor[1] <= '9')
  {
    return false;
  }

  unsigned int prefix = 0;
  while (*cursor >= '0' && *cursor <= '9')
  {
    prefix = prefix * 10U + (unsigned int) (*cursor - '0');
    if (prefix > 32U)
    {
      return false;
    }
    cursor++;
  }
  if (*cursor != '\0')
  {
    return false;
  }

  uint32_t parsed_address;
  if (!ip_parse(address, &parsed_address))
  {
    return false;
  }

  block->addr = parsed_address;
  block->prefix = (uint8_t) prefix;
  return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN])
{
  char address[IP_STR_LEN];
  ip_format(block.addr, address);
  unsigned int prefix = block.prefix > 32U ? 32U : (unsigned int) block.prefix;
  (void) snprintf(out, CIDR_STR_LEN, "%s/%u", address, prefix);
}

uint32_t cidr_mask(uint8_t prefix)
{
  if (prefix == 0)
  {
    return 0;
  }
  if (prefix > 32)
  {
    prefix = 32;
  }
  return UINT32_MAX << (32U - prefix);
}

uint32_t cidr_network(cidr_t block)
{
  return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block)
{
  uint32_t mask = cidr_mask(block.prefix);
  return cidr_network(block) | ~mask;
}

uint32_t cidr_first_host(cidr_t block)
{
  uint32_t network = cidr_network(block);
  return block.prefix <= 30 ? network + 1U : network;
}

uint32_t cidr_last_host(cidr_t block)
{
  uint32_t broadcast = cidr_broadcast(block);
  return block.prefix <= 30 ? broadcast - 1U : broadcast;
}

uint64_t cidr_host_count(cidr_t block)
{
  unsigned int prefix = block.prefix > 32 ? 32U : (unsigned int) block.prefix;
  if (prefix == 32)
  {
    return 1;
  }
  if (prefix == 31)
  {
    return 2;
  }
  return (UINT64_C(1) << (32U - prefix)) - 2U;
}

bool cidr_contains(cidr_t block, uint32_t addr)
{
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == cidr_network(block);
}

bool cidr_is_network(cidr_t block)
{
  return block.addr == cidr_network(block);
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
  if (out == NULL || block.prefix > 32 || !cidr_is_network(block)
      || n == 0 || (n & (n - 1U)) != 0 || i >= n)
  {
    return false;
  }

  unsigned int borrowed = 0;
  for (uint32_t parts = n; parts > 1; parts >>= 1)
  {
    borrowed++;
  }
  unsigned int new_prefix = (unsigned int) block.prefix + borrowed;
  if (new_prefix > 32)
  {
    return false;
  }

  unsigned int host_bits = 32U - new_prefix;
  uint64_t offset = (uint64_t) i << host_bits;
  cidr_t subnet = {
    block.addr + (uint32_t) offset,
    (uint8_t) new_prefix
  };
  *out = subnet;
  return true;
}

bool route_parse(const char *line, route_t *route)
{
  if (line == NULL || route == NULL)
  {
    return false;
  }

  const char *cursor = line;
  const char *fields[3];
  size_t lengths[3];

  for (size_t field = 0; field < 3; field++)
  {
    while (*cursor == ' ' || *cursor == '\t')
    {
      cursor++;
    }
    if (*cursor == '\0' || *cursor == '\n' || *cursor == '\r')
    {
      return false;
    }

    fields[field] = cursor;
    while (*cursor != '\0' && *cursor != ' ' && *cursor != '\t'
           && *cursor != '\n' && *cursor != '\r')
    {
      cursor++;
    }
    lengths[field] = (size_t) (cursor - fields[field]);
  }

  while (*cursor == ' ' || *cursor == '\t' || *cursor == '\n'
         || *cursor == '\r')
  {
    cursor++;
  }
  if (*cursor != '\0' || lengths[2] >= DEV_NAME_LEN)
  {
    return false;
  }

  route_t parsed = { 0 };
  if (lengths[0] == sizeof("default") - 1
      && memcmp(fields[0], "default", sizeof("default") - 1) == 0)
  {
    parsed.dest.addr = 0;
    parsed.dest.prefix = 0;
  }
  else
  {
    char destination[CIDR_STR_LEN];
    if (lengths[0] >= sizeof(destination))
    {
      return false;
    }
    memcpy(destination, fields[0], lengths[0]);
    destination[lengths[0]] = '\0';
    if (!cidr_parse(destination, &parsed.dest)
        || !cidr_is_network(parsed.dest))
    {
      return false;
    }
  }

  if (lengths[1] == sizeof("direct") - 1
      && memcmp(fields[1], "direct", sizeof("direct") - 1) == 0)
  {
    parsed.gateway = 0;
  }
  else
  {
    char gateway[IP_STR_LEN];
    if (lengths[1] >= sizeof(gateway))
    {
      return false;
    }
    memcpy(gateway, fields[1], lengths[1]);
    gateway[lengths[1]] = '\0';
    if (!ip_parse(gateway, &parsed.gateway))
    {
      return false;
    }
  }

  memcpy(parsed.dev, fields[2], lengths[2]);
  parsed.dev[lengths[2]] = '\0';
  *route = parsed;
  return true;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
  if (table == NULL || route == NULL || route->dest.prefix > 32
      || !cidr_is_network(route->dest))
  {
    return false;
  }

  size_t new_capacity = table->capacity;
  if (table->count == table->capacity)
  {
    size_t max_capacity = SIZE_MAX / sizeof(*table->routes);
    if (table->capacity == 0)
    {
      new_capacity = 8;
    }
    else
    {
      if (table->capacity > max_capacity / 2)
      {
        return false;
      }
      new_capacity = table->capacity * 2;
    }
  }

  for (size_t i = 0; i < table->count; i++)
  {
    if (table->routes[i].dest.addr == route->dest.addr
        && table->routes[i].dest.prefix == route->dest.prefix)
    {
      return false;
    }
  }

  if (table->count == table->capacity)
  {
    route_t *routes = realloc(table->routes, new_capacity * sizeof(*routes));
    if (routes == NULL) // GCOVR_EXCL_BR_LINE
    {
      return false; // GCOVR_EXCL_LINE
    }
    table->routes = routes;
    table->capacity = new_capacity;
  }

  table->routes[table->count++] = *route;
  return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
  if (bad_line != NULL)
  {
    *bad_line = 0;
  }
  if (table == NULL || in == NULL)
  {
    return false;
  }

  char *line = NULL;
  size_t capacity = 0;
  size_t line_number = 0;
  ssize_t line_length;
  while ((line_length = getline(&line, &capacity, in)) != -1)
  {
    line_number++;
    char *comment = strchr(line, '#');
    if (comment != NULL)
    {
      *comment = '\0';
    }

    const char *content = line;
    while (*content == ' ' || *content == '\t' || *content == '\n'
           || *content == '\r')
    {
      content++;
    }
    if (*content != '\0')
    {
      route_t route;
      if (!route_parse(content, &route) || !table_add(table, &route))
      {
        if (bad_line != NULL)
        {
          *bad_line = line_number;
        }
        free(line);
        return false;
      }
    }
  }

  free(line);
  if (ferror(in)) // GCOVR_EXCL_BR_LINE
  {
    return false; // GCOVR_EXCL_LINE
  }
  return feof(in) != 0; // GCOVR_EXCL_BR_LINE
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#include "lab.h"
#include <errno.h>
#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#ifdef TEST
#define main main_exclude

static void print_usage(FILE *stream)
{
  (void) fprintf(stream,
                 "Usage: myapp info <address>/<prefix>\n"
                 "       myapp split <network>/<prefix> <count>\n"
                 "       myapp lookup <table-file> <address>...\n");
}

static bool parse_count(const char *text, uint32_t *count)
{
  if (text == NULL || count == NULL || *text == '\0')
  {
    return false;
  }

  uint32_t parsed = 0;
  const unsigned char *cursor = (const unsigned char *) text;
  while (*cursor >= '0' && *cursor <= '9')
  {
    uint32_t digit = (uint32_t) (*cursor - '0');
    if (parsed > (UINT32_MAX - digit) / 10U)
    {
      return false;
    }
    parsed = parsed * 10U + digit;
    cursor++;
  }
  if (*cursor != '\0')
  {
    return false;
  }

  *count = parsed;
  return true;
}

static int run_info(const char *text)
{
  cidr_t block;
  if (!cidr_parse(text, &block))
  {
    (void) fprintf(stderr, "myapp: %s: not a valid address/prefix\n", text);
    return 1;
  }

  char address[IP_STR_LEN];
  char mask[IP_STR_LEN];
  char network[IP_STR_LEN];
  char broadcast[IP_STR_LEN];
  char first_host[IP_STR_LEN];
  char last_host[IP_STR_LEN];
  ip_format(block.addr, address);
  ip_format(cidr_mask(block.prefix), mask);
  ip_format(cidr_network(block), network);
  ip_format(cidr_broadcast(block), broadcast);
  ip_format(cidr_first_host(block), first_host);
  ip_format(cidr_last_host(block), last_host);

  (void) printf("address    %s\n"
                "prefix     /%u\n"
                "netmask    %s\n"
                "network    %s\n"
                "broadcast  %s\n"
                "first host %s\n"
                "last host  %s\n"
                "hosts      %" PRIu64 "\n",
                address, (unsigned int) block.prefix, mask, network,
                broadcast, first_host, last_host, cidr_host_count(block));
  return 0;
}

static int run_split(const char *text, const char *count_text)
{
  cidr_t block;
  if (!cidr_parse(text, &block))
  {
    (void) fprintf(stderr, "myapp: %s: not a valid network/prefix\n", text);
    return 1;
  }

  uint32_t count;
  if (!parse_count(count_text, &count) || count == 0
      || (count & (count - 1U)) != 0)
  {
    (void) fprintf(stderr, "myapp: %s: count must be a positive power of two\n",
                   count_text);
    return 1;
  }
  if (!cidr_is_network(block))
  {
    (void) fprintf(stderr, "myapp: %s: block has host bits set\n", text);
    return 1;
  }

  for (uint32_t i = 0; i < count; i++)
  {
    cidr_t subnet;
    if (!cidr_subnet(block, count, i, &subnet))
    {
      (void) fprintf(stderr,
                     "myapp: cannot split %s into %" PRIu32 " subnets\n",
                     text, count);
      return 1;
    }
    char formatted[CIDR_STR_LEN];
    cidr_format(subnet, formatted);
    (void) puts(formatted);
  }
  return 0;
}

static int run_lookup(const char *path, int address_count, char *const addresses[])
{
  FILE *input = fopen(path, "r");
  if (input == NULL)
  {
    (void) fprintf(stderr, "myapp: %s: %s\n", path, strerror(errno));
    return 2;
  }

  fwd_table_t *table = table_create();
  if (table == NULL)
  {
    (void) fprintf(stderr, "myapp: could not allocate forwarding table\n");
    (void) fclose(input);
    return 2;
  }

  size_t bad_line = 0;
  if (!table_load(table, input, &bad_line))
  {
    if (bad_line != 0)
    {
      (void) fprintf(stderr, "myapp: %s:%zu: not a valid route\n",
                     path, bad_line);
    }
    else
    {
      (void) fprintf(stderr, "myapp: %s: could not read table\n", path);
    }
    table_destroy(table);
    (void) fclose(input);
    return 2;
  }
  (void) fclose(input);

  int exit_status = 0;
  for (int i = 0; i < address_count; i++)
  {
    uint32_t address;
    if (!ip_parse(addresses[i], &address))
    {
      (void) fprintf(stderr, "myapp: %s: not a valid address\n", addresses[i]);
      exit_status = 1;
      continue;
    }

    char formatted_address[IP_STR_LEN];
    ip_format(address, formatted_address);
    const route_t *route = table_lookup(table, address);
    if (route == NULL)
    {
      (void) printf("%s unreachable\n", formatted_address);
      if (exit_status == 0)
      {
        exit_status = 3;
      }
    }
    else if (route->gateway == 0)
    {
      (void) printf("%s dev %s\n", formatted_address, route->dev);
    }
    else
    {
      char gateway[IP_STR_LEN];
      ip_format(route->gateway, gateway);
      (void) printf("%s via %s dev %s\n",
                    formatted_address, gateway, route->dev);
    }
  }

  table_destroy(table);
  return exit_status;
}

int main(int argc, char *argv[])
{
  if (argc == 1)
  {
    print_usage(stdout);
    return 0;
  }

  if (strcmp(argv[1], "info") == 0)
  {
    if (argc != 3)
    {
      print_usage(stderr);
      return 1;
    }
    return run_info(argv[2]);
  }
  if (strcmp(argv[1], "split") == 0)
  {
    if (argc != 4)
    {
      print_usage(stderr);
      return 1;
    }
    return run_split(argv[2], argv[3]);
  }
  if (strcmp(argv[1], "lookup") == 0)
  {
    if (argc < 4)
    {
      print_usage(stderr);
      return 1;
    }
    return run_lookup(argv[2], argc - 3, &argv[3]);
  }

  (void) fprintf(stderr, "myapp: unknown command: %s\n", argv[1]);
  print_usage(stderr);
  return 1;
}
